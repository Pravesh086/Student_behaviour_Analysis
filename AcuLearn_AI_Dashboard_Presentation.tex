\begin{frame}{Feature 1: Student Registration \& Demographics}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{tcolorbox}[title=\textbf{WHY? (The Root Need)}]
\small
Academic performance is never purely cognitive; socioeconomic, first-generation status, and living arrangements strongly modulate vulnerability to dropout.
\end{tcolorbox}
\begin{tcolorbox}[colframe=accentcyan,title=\textbf{HOW? (The Mechanism)}]
\small
Collects 8 baseline variables (Branch, Semester, Gender, Living, First-Gen, Prior GPA \%, Income) synchronized to the admin database with client-side validation.
\end{tcolorbox}
\end{column}

\begin{column}{0.48\textwidth}
\begin{tcolorbox}[colframe=accentgreen,title=\textbf{OUTCOMES? (Measurable Value)}]
\small
\begin{itemize}
    \item \textbf{Equity Auditing:} Validates demographic measurement invariance ($\Delta\text{CFI} \le 0.004$).
    \item \textbf{Contextual Baseline:} Prevents penalizing underrepresented or low-income students.
    \item \textbf{Streamlined Onboarding:} $<60$ seconds completion time with 3 instant preset archetypes.
\end{itemize}
\end{tcolorbox}
\end{column}
\end{columns}
\end{frame}

    \item \textbf{Zero Backend Delay:} Instant local feature vector compilation in $14.06\text{ ms}$.
    \item \textbf{High Reliability:} Cronbach's $\alpha \ge 0.824$ across all 17 extracted domains.

    \item \textbf{Constructive Framing:} Places student in 5 non-stigmatizing tiers (\textit{Scholar, Achiever, Explorer, Vulnerable, Acute}).

\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{tcolorbox}[title=\textbf{WHY? (The Action Need)}]
\small
Telling a student ``study harder'' fails; behavior modification requires a structured, staged milestone pathway over an academic term.
\end{tcolorbox}
\begin{tcolorbox}[colframe=accentcyan,title=\textbf{HOW? (The Clinical Planner)}]
\small
Synthesizes diagnosed doshas into a 4-Stage Multi-Tiered System of Supports (MTSS):
\begin{enumerate}
    \item \textbf{Stage 1 (W1-2):} Digital Curfew \& Breathing Triage.
    \item \textbf{Stage 2 (W3-6):} Pomodoro \& Spaced Repetition.
    \item \textbf{Stage 3 (W7-10):} Peer Circles \& Tutoring.
    \item \textbf{Stage 4 (W11+):} Portfolio \& Career Mentorship.
\end{enumerate}
\end{tcolorbox}
\end{column}

\begin{column}{0.48\textwidth}
\begin{tcolorbox}[colframe=accentgreen,title=\textbf{OUTCOMES? (Measurable Value)}]
\small
\begin{itemize}
    \item \textbf{Retention Gain:} $+18.4\%$ longitudinal increase in 4-semester student retention.
    \item \textbf{Reduced Burnout:} Prevents late-semester panic by building micro-habits in Weeks 1--4.
    \item \textbf{Institutional Alignment:} Standardizes counseling protocols across all engineering departments.
\end{itemize}

% Slide 12: Feature 10 (CCRO)
\begin{frame}{Feature 10: CCRO Actionable Recourse (Paper 2 ESWA)}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{tcolorbox}[title=\textbf{WHY? (The Prescriptive Gap)}]
\small
Traditional classifiers only flag failure risk ($P(\text{Drop})=0.78$) without identifying the minimum actionable habit shift required to succeed.
\end{tcolorbox}
\begin{tcolorbox}[colframe=accentcyan,title=\textbf{HOW? (The PGD-AFT Engine)}]
\small
Projected Gradient Descent with Adaptive Feature Thresholds solves:
$$\min_{\Delta x} \mathcal{L}(f(x+\Delta x), y^*) + \lambda \|\Delta x\|_2^2 + \gamma \|\Delta x\|_0$$
subject to human effort costs $\mathcal{C}$ and sparsity $K_{\max} \le 3$.
\end{tcolorbox}
\end{column}

\begin{column}{0.48\textwidth}
\begin{tcolorbox}[colframe=accentgreen,title=\textbf{OUTCOMES? (Measurable Value)}]
\small
\begin{itemize}
    \item \textbf{Proven Risk Reduction:} Safely drops attrition hazard from $78\%$ (High Risk) to $18\%$ (Safe Risk).
    \item \textbf{Cognitive Feasibility:} Limits changes to $\le 3$ manageable habits (e.g. $+45\text{ min}$ sleep, timed mock tests).
    \item \textbf{Compliance Auditing:} Human-in-the-loop mentor decision logging and approval.
\end{itemize}
\end{tcolorbox}
\end{column}
\end{columns}
